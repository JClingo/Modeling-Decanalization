\newcommand{\panel}[5]{%
  \filldraw[fill=black!2,draw=rule,rounded corners=4pt] (#1,#2-0.1) rectangle (#1+\pw,#2+2.5);
  \fill[black!6] plot[domain=0:1,samples=90,smooth,variable=\u]
    ({#1+0.1+(\pw-0.2)*\u},{#2+\rim+\dep*land(\u,#3,#4)}) -- (#1+\pw-0.1,#2) -- (#1+0.1,#2) -- cycle;
  \draw[#5,line width=1.1pt] plot[domain=0:1,samples=90,smooth,variable=\u]
    ({#1+0.1+(\pw-0.2)*\u},{#2+\rim+\dep*land(\u,#3,#4)});}
% \ball{x}{y}{u}{familiar depth}{new depth}
\newcommand{\ball}[5]{%
  \pgfmathsetmacro{\bx}{#1+0.1+(\pw-0.2)*#3}
  \pgfmathsetmacro{\by}{#2+\rim+\dep*land(#3,#4,#5)+0.12}
  \filldraw[fill=ink,draw=white,line width=0.6pt] (\bx,\by) circle (0.12);}
